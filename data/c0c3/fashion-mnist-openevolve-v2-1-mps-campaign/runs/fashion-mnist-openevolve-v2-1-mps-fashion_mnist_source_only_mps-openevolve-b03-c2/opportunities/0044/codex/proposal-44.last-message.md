def training_loss(
    model: nn.Module,
    images: torch.Tensor,
    labels: torch.Tensor,
    step: int,
    total_steps: int,
) -> torch.Tensor:
    del step, total_steps
    paired_images = torch.cat((images, images.flip(-1)), dim=0)
    paired_labels = torch.cat((labels, labels), dim=0)
    logits = model(paired_images)

    classification = F.cross_entropy(
        logits,
        paired_labels,
        label_smoothing=0.02,
    )

    original_logits, mirrored_logits = logits.chunk(2, dim=0)
    original_log_probs = F.log_softmax(original_logits, dim=1)
    mirrored_log_probs = F.log_softmax(mirrored_logits, dim=1)
    original_probs = original_log_probs.exp()
    mirrored_probs = mirrored_log_probs.exp()
    mean_probs = 0.5 * (original_probs + mirrored_probs)
    mean_log_probs = mean_probs.clamp_min(1.0e-7).log()

    consistency = 0.5 * (
        (
            original_probs
            * (original_log_probs - mean_log_probs)
        ).sum(dim=1).mean()
        + (
            mirrored_probs
            * (mirrored_log_probs - mean_log_probs)
        ).sum(dim=1).mean()
    )
    return classification + 0.5 * consistency
